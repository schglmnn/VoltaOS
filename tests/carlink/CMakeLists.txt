cmake_minimum_required(VERSION 3.16)
project(carlink_video_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Multimedia Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec libavutil)
enable_testing()
add_executable(carlink_video_test video_test.cpp ../../src/carlink/CarVideoDecoder.cpp)
target_include_directories(carlink_video_test PRIVATE ../../src/carlink)
target_link_libraries(carlink_video_test PRIVATE Qt6::Core Qt6::Multimedia Qt6::Test PkgConfig::AVCODEC)
add_test(NAME carlink_video_recovery COMMAND carlink_video_test)
