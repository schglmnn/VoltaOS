cmake_minimum_required(VERSION 3.16)
project(map_gl_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick Test)
enable_testing()
add_executable(map_gl_test map_gl_test.cpp)
target_link_libraries(map_gl_test PRIVATE Qt6::Quick Qt6::Test)
target_compile_definitions(map_gl_test PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME map_shared_gl_state COMMAND map_gl_test)
