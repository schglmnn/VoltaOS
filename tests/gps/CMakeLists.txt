cmake_minimum_required(VERSION 3.16)
project(gps_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
enable_testing()

target_include_directories(gps_test PRIVATE ../../src/nav)
target_link_libraries(gps_test PRIVATE Qt6::Core Qt6::Test)
add_test(NAME gps_simulation COMMAND gps_test)
